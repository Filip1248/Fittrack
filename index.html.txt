                select.innerHTML = `<option value="">Žiadne cviky pre túto kategóriu</option>`;
                return;
            }

            select.innerHTML = filtered.map(ex => `
                <option value="${ex.id}">${ex.name} (${ex.muscle})</option>
            `).join('');
        }

            container.innerHTML = templates.map(tpl => `

        function loadTemplate(tplId) {
            const tpl = templates.find(t => t.id === tplId);
            if (!tpl) return;

            activeTemplateId = tplId;
            document.getElementById('activeTemplateTitle').textContent = `Aktívny tréning: ${tpl.name}`;
            document.getElementById('activeTemplateBar').classList.remove('hidden');

            if (tpl.exercises.length > 0) {
                const firstExId = tpl.exercises[0];
                const foundEx = exercises.find(e => e.id === firstExId);
                if (foundEx) {
                    activeMuscleFilter = 'Všetky';
                    renderMuscleTabs();
                    renderExerciseOptions();
                    document.getElementById('exerciseSelect').value = firstExId;
                    onExerciseChange();
                }
            }
        }

        function handleSaveTemplate(e) {
            e.preventDefault();
            const name = document.getElementById('templateNameInput').value.trim();
            const checkboxes = document.querySelectorAll('#templateExerciseChecklist input[type="checkbox"]:checked');
            const selectedExIds = Array.from(checkboxes).map(cb => cb.value);

            if (!name || selectedExIds.length === 0) return;

            const newTpl = {
                id: 'tpl_' + Date.now(),
                name: name,
                exercises: selectedExIds
            };

        // --- EXERCISE SELECTION & PROGRESSIVE OVERLOAD PREVIEW ---
        function onExerciseChange() {
            const selectedExId = document.getElementById('exerciseSelect').value;
            if (!selectedExId) return;

            const today = new Date().toISOString().split('T')[0];
            const todaySetsForEx = workoutLogs.filter(l => l.date === today && l.exerciseId === selectedExId);
            
            document.getElementById('nextSetNumber').textContent = todaySetsForEx.length + 1;

            // Find last performance before today
            const prevLogsForEx = workoutLogs.filter(l => l.exerciseId === selectedExId && l.date !== today);
            
            if (prevLogsForEx.length > 0) {
                // Get absolute max weight recorded previously
                const bestPrevSet = prevLogsForEx.reduce((prev, current) => (Number(prev.weight) > Number(current.weight)) ? prev : current);
                
                document.getElementById('prevPerformanceText').textContent = `${bestPrevSet.weight} kg x ${bestPrevSet.reps} reps (RPE ${bestPrevSet.rpe})`;
                document.getElementById('prevPerformanceCard').classList.remove('hidden');
                document.getElementById('prevPerformanceCard').dataset.weight = bestPrevSet.weight;
                document.getElementById('prevPerformanceCard').dataset.reps = bestPrevSet.reps;
            } else {
                document.getElementById('prevPerformanceCard').classList.add('hidden');
            }
        }

        function applyPrevPerformance() {
            const card = document.getElementById('prevPerformanceCard');
            if (card.dataset.weight) {
                document.getElementById('inputWeight').value = card.dataset.weight;
                document.getElementById('inputReps').value = card.dataset.reps;
            }
        }

        // --- RPE SLIDER INTERACTION ---
        function updateRpeDisplay(val) {
            document.getElementById('rpeValueDisplay').textContent = val;
            const subtext = document.getElementById('rpeSubtext');
            const rpeNum = parseFloat(val);

            if (rpeNum === 10) subtext.textContent = '0 reps v rezerve (Zlyhanie)';
            else if (rpeNum === 9.5) subtext.textContent = 'Možno ešte 1 rep';
            else if (rpeNum === 9) subtext.textContent = '1 rep v rezerve';
            else if (rpeNum === 8.5) subtext.textContent = '1-2 reps v rezerve';
            else if (rpeNum === 8) subtext.textContent = '2 reps v rezerve';
            else if (rpeNum === 7) subtext.textContent = '3 reps v rezerve';
            else subtext.textContent = 'Rozcvička';
        }

        // --- SUBMIT LOGGED SET ---
        function handleSetSubmit(e) {
            e.preventDefault();

            const exerciseId = document.getElementById('exerciseSelect').value;
            const weight = parseFloat(document.getElementById('inputWeight').value);
            const reps = parseInt(document.getElementById('inputReps').value);
            const rpe = parseFloat(document.getElementById('inputRpe').value);

            if (!exerciseId || isNaN(weight) || isNaN(reps)) return;

            const selectedExObj = exercises.find(e => e.id === exerciseId);
            const today = new Date().toISOString().split('T')[0];

            const todaySetsForEx = workoutLogs.filter(l => l.date === today && l.exerciseId === exerciseId);
            const setNum = todaySetsForEx.length + 1;

            const newLog = {
                id: 'log_' + Date.now(),
                date: today,
                timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
                exerciseId: exerciseId,
                exerciseName: selectedExObj ? selectedExObj.name : 'Unknown',
                muscle: selectedExObj ? selectedExObj.muscle : 'Other',
                setNumber: setNum,
                weight: weight,
                reps: reps,
                rpe: rpe
            };

            workoutLogs.unshift(newLog);
            localStorage.setItem('fitTrack_logs', JSON.stringify(workoutLogs));

            renderTodaySets();
            onExerciseChange();

            // Auto start rest timer
            startTimer();
            toggleTimerModal();
        }

        // --- REST TIMER CONTROLS ---
        function toggleTimerModal() {
            const modal = document.getElementById('timerModal');
            modal.classList.toggle('hidden');
        }

        function setTimerDuration(seconds) {
            timerSeconds = seconds;
            initialTimerSeconds = seconds;
            updateTimerDisplay();
            if (isTimerRunning) {
                clearInterval(timerInterval);
                startTimer();
            }
        }

        function startTimer() {
            if (isTimerRunning) clearInterval(timerInterval);
            isTimerRunning = true;
            document.getElementById('timerToggleBtn').textContent = 'Pauza';

            timerInterval = setInterval(() => {
                if (timerSeconds > 0) {
                    timerSeconds--;
                    updateTimerDisplay();
                } else {
                    clearInterval(timerInterval);
                    isTimerRunning = false;
                    playBeep();
                    document.getElementById('timerToggleBtn').textContent = 'Štart';
                }
            }, 1000);
        }

        function toggleTimerState() {
            if (isTimerRunning) {
                clearInterval(timerInterval);
                isTimerRunning = false;
                document.getElementById('timerToggleBtn').textContent = 'Pokračovať';
            } else {
                startTimer();
            }
        }

        function resetTimer() {
            clearInterval(timerInterval);
            isTimerRunning = false;
            timerSeconds = initialTimerSeconds;
            updateTimerDisplay();
            document.getElementById('timerToggleBtn').textContent = 'Štart';
        }

        function updateTimerDisplay() {
            const mins = Math.floor(timerSeconds / 60);
            const secs = timerSeconds % 60;
            const str = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
            document.getElementById('headerTimerDisplay').textContent = str;
            document.getElementById('modalTimerDisplay').textContent = str;
        }

        function playBeep() {
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(880, ctx.currentTime);
                osc.connect(ctx.destination);
                osc.start();
                osc.stop(ctx.currentTime + 0.5);
            } catch(e) { console.log('Audio not supported'); }
        }
